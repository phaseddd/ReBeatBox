```java
/**
 * Wires Radiance Timeline hover border-color and press scale animations
 * onto a JButton. Per D-05, D-06, D-07.
 *
 * <p>Hover: borderColor transitions from BORDER_IDLE to accentForHue(HUE_TRANSPORT)
 * in 200ms. Press: scale drops to 0.95x in 75ms, springs back to 1.00x in 150ms.
 */
private void wireButtonAnimation(JButton button, float regionHue) {
    Color accentColor = ThemeManager.accentForHue(regionHue);

    // Hover enter animation
    button.addMouseListener(new MouseAdapter() {
        private Timeline hoverTimeline;
        private Timeline pressTimeline;

        @Override
        public void mouseEntered(MouseEvent e) {
            if (hoverTimeline != null) hoverTimeline.abort();
            hoverTimeline = Timeline.builder(button)
                .addPropertyToInterpolate("borderColor",
                    ThemeManager.BORDER_IDLE, accentColor)
                .setDuration(200)
                .setEase(new SplineEase(0.4f, 0.0f, 0.2f, 1.0f))
                .build();
            hoverTimeline.play();
        }

        @Override
        public void mouseExited(MouseEvent e) {
            if (hoverTimeline != null) hoverTimeline.abort();
            hoverTimeline = Timeline.builder(button)
                .addPropertyToInterpolate("borderColor",
                    accentColor, ThemeManager.BORDER_IDLE)
                .setDuration(250)
                .setEase(new SplineEase(0.4f, 0.0f, 0.2f, 1.0f))
                .build();
            hoverTimeline.play();
        }

        @Override
        public void mousePressed(MouseEvent e) {
            if (SwingUtilities.isLeftMouseButton(e)) {
                if (pressTimeline != null) pressTimeline.abort();
                pressTimeline = Timeline.builder(button)
                    .addPropertyToInterpolate("scale", 1.00f, 0.95f)
                    .setDuration(75)
                    .build();
                pressTimeline.play();
            }
        }

        @Override
        public void mouseReleased(MouseEvent e) {
            if (SwingUtilities.isLeftMouseButton(e)) {
                if (pressTimeline != null) pressTimeline.abort();
                pressTimeline = Timeline.builder(button)
                    .addPropertyToInterpolate("scale", 0.95f, 1.00f)
                    .setDuration(150)
                    .setEase(new SplineEase(0.4f, 0.0f, 0.2f, 1.0f))
                    .build();
                pressTimeline.play();
            }
        }
    });
}
```

```java
private void toggle(ActionEvent e) {
    if (!expanded) {
        // Expanding: capture snapshot when collapsed (0px wide = empty), so resize first
        // Actually, SidebarPanel is 0px wide when collapsed — can't snapshot.
        // Strategy: resize to expanded width, THEN apply glitch.
        // For a 150ms glitch per D-18:
        setPreferredSize(new Dimension(expandedWidth, getHeight()));
        revalidate();
        // No snapshot needed for expand — the component appears after revalidate.
        // The glitch effect applies to the newly visible content.
    } else {
        // Collapsing: capture current content, apply glitch, then collapse
        performGlitchCollapse();
        return; // glitch handles the collapse
    }

    expanded = !expanded;
    updateToggleIcon();
}

private void performGlitchCollapse() {
    int w = getWidth();
    int h = getHeight();
    if (w <= 0 || h <= 0) {
        // Fallback: instant collapse if no pixels to snapshot
        expanded = false;
        updateToggleIcon();
        setPreferredSize(new Dimension(collapsedWidth, getHeight()));
        revalidate();
        repaint();
        return;
    }

    // Capture snapshot of current sidebar content
    BufferedImage snapshot = new BufferedImage(w, h, BufferedImage.TYPE_INT_ARGB);
    Graphics2D g2 = snapshot.createGraphics();
    paint(g2); // draw current state
    g2.dispose();

    // Create glitch transition with 150ms duration (D-18)
    // Max offset: 15px red left, 15px blue right (UI-SPEC)
    final int maxOffset = 15;

    Timeline glitchTimeline = Timeline.builder(this)
        .setDuration(150)
        .addCallback(new TimelineCallback() {
            @Override
            public void onTimelinePulse(float durationFraction, float timelinePosition) {
                // Bell curve: peak at 0.5, returns to 0 at start/end
                float bellCurve = 4.0f * timelinePosition * (1.0f - timelinePosition);
                int offset = Math.round(maxOffset * bellCurve);

                BufferedImage glitched = GlitchTransition.applyRgbSplit(snapshot, -offset, offset);
                // Paint glitched snapshot directly (override temporarily)
                // Store reference for paintComponent to use
                setGlitchImage(glitched);
                repaint();
            }

            @Override
            public void onTimelineStateChanged(TimelineState oldState, TimelineState newState,
                                               float durationFraction, float timelinePosition) {
                if (newState == TimelineState.DONE) {
                    setGlitchImage(null);
                    expanded = false;
                    updateToggleIcon();
                    setPreferredSize(new Dimension(collapsedWidth, getHeight()));
                    revalidate();
                    repaint();
                    snapshot.flush();
                }
            }
        })
        .build();
    glitchTimeline.play();
}

/** Current glitch frame image; null when no transition active. */
private transient BufferedImage glitchImage;

private void setGlitchImage(BufferedImage img) {
    this.glitchImage = img;
}

@Override
protected void paintComponent(Graphics g) {
    if (glitchImage != null) {
        g.drawImage(glitchImage, 0, 0, null);
    } else {
        super.paintComponent(g);
    }
}

private void updateToggleIcon() {
    String iconName = expanded ? "collapse-left" : "expand-right";
    String tooltip = expanded ? "Collapse sidebar" : "Expand sidebar";
    toggleButton.setToolTipText(tooltip);
    toggleButton.getAccessibleContext().setAccessibleName(tooltip);
    try {
        BufferedImage icon = SvgIconLoader.getIcon(iconName, 24);
        if (icon != null) toggleButton.setIcon(new ImageIcon(icon));
    } catch (Exception e) {
        System.err.println("Failed to load sidebar icon: " + e.getMessage());
    }
}
```
